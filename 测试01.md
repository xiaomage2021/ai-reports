### 1. 修复 P0-05-A & P0-05-B：出站调度主管道与安全跃迁（严格异常白名单 + 租约绝对时间校验）

```python
import secrets
import time
import json
import asyncio
import logging

# 显式引入 Telethon / MTProto 确定性未产生的 API 异常类型
from telethon.errors import (
    ChatNotFound, UserDeactivated, FloodWait, ChannelPrivate,
    ChatRestricted, UserBlocked, BadRequestError
)

logger = logging.getLogger("tg-monitor-outbound")

# 【P0-05-A 修复】：确定性未产生外部推送的安全异常白名单
# 只有命中名单的异常，才证明外部绝未收到，允许进入 RETRY_WAIT 退避重发
DETERMINISTIC_FAILED_EXCEPTIONS = (
    ChatNotFound, UserDeactivated, FloodWait, ChannelPrivate,
    ChatRestricted, UserBlocked, BadRequestError
)

async def transition_to_sending(db, write_lock: asyncio.Lock, task_id: str, worker_id: str, claim_token: str) -> bool:
    """
    【事务 4.2 修订：调用外部 Telegram API 前的状态跃迁】
    【P0-05-B 修复】：必须严格核验 claim_expires_at >= now，防超时滞后 Worker 越权调用
    """
    now = int(time.time())
    send_expires_at = now + 60  # 发送中租约 60 秒
    async with write_lock:
        await db.execute("BEGIN IMMEDIATE")
        try:
            cur = await db.execute(
                "UPDATE outbound_queue SET status = 'SENDING', send_started_at = ?, claim_expires_at = ?, updated_at = ? "
                "WHERE task_id = ? AND worker_id = ? AND claim_token = ? AND status = 'CLAIMED' AND claim_expires_at >= ?",
                (now, send_expires_at, now, task_id, worker_id, claim_token, now)
            )
            if cur.rowcount == 1:
                await db.execute("COMMIT")
                return True
            await db.execute("ROLLBACK")
            return False
        except Exception:
            await db.execute("ROLLBACK")
            return False

async def dispatch_outbound_task(bot_client, db, write_lock: asyncio.Lock, worker_id: str, task: dict):
    """
    【事务 4.3 修订：出站调度主管道】
    【P0-05-A 修复】：杜绝粗暴 except Exception！非白名单异常一律按 UNKNOWN_ERROR 处理，阻断盲目重发
    """
    task_id = task["task_id"]
    claim_token = task["claim_token"]
    target_group_id = task["target_group_id"]
    payload = task["payload"]

    # 1. 跃迁核验（带有效期核验）
    if not await transition_to_sending(db, write_lock, task_id, worker_id, claim_token):
        logger.warning(f"[出站调度] 任务 {task_id} 跃迁至 SENDING 失败（租约已过期或被回收），终止调用！")
        return

    # 2. 调用 Telegram API 发送
    now = int(time.time())
    telegram_msg_id = None
    explicit_failure = False
    is_unknown_network_error = False
    error_detail = None

    try:
        msg = await asyncio.wait_for(
            bot_client.send_message(
                target_group_id,
                payload["text"],
                buttons=payload.get("buttons"),
                link_preview=False
            ),
            timeout=10.0
        )
        telegram_msg_id = msg.id
    except DETERMINISTIC_FAILED_EXCEPTIONS as api_err:
        # 场景 B：命中白名单的明确报错（明确未产生外部推送，允许安全退避重发）
        explicit_failure = True
        error_detail = f"API明确报错(未发送): {api_err}"
    except Exception as unknown_err:
        # 场景 C：网络超时、断连、响应解析失败、RPC未知异常等（结果不确定，绝对禁止盲目重发！）
        is_unknown_network_error = True
        error_detail = f"不确定异常(需对账): {unknown_err}"

    # 3. 状态原子落库
    async with write_lock:
        await db.execute("BEGIN IMMEDIATE")
        try:
            if telegram_msg_id is not None:
                # 明确成功 -> SENT
                cur = await db.execute(
                    "UPDATE outbound_queue SET status = 'SENT', telegram_msg_id = ?, sent_at = ?, updated_at = ? "
                    "WHERE task_id = ? AND worker_id = ? AND claim_token = ? AND status = 'SENDING'",
                    (telegram_msg_id, now, now, task_id, worker_id, claim_token)
                )
                if cur.rowcount == 1:
                    await db.execute("COMMIT")
                    logger.info(f"[出站推送成功] 任务 {task_id} 已送达，msg_id={telegram_msg_id}")
                    return
            elif explicit_failure:
                # 明确未发失败 -> RETRY_WAIT (退避重试)
                cur = await db.execute(
                    "UPDATE outbound_queue SET status = 'RETRY_WAIT', worker_id = NULL, claim_token = NULL, "
                    "retry_count = retry_count + 1, next_retry_at = ? + (5 * (1 << retry_count)), last_error = ?, updated_at = ? "
                    "WHERE task_id = ? AND worker_id = ? AND claim_token = ? AND status = 'SENDING'",
                    (now, error_detail, task_id, worker_id, claim_token)
                )
                if cur.rowcount == 1:
                    await db.execute("COMMIT")
                    logger.warning(f"[出站明确失败] 任务 {task_id} 进入 RETRY_WAIT: {error_detail}")
                    return
            else:
                # 结果未知 -> 强制进入 UNKNOWN_ERROR 等待 MTProto 三次对账
                cur = await db.execute(
                    "UPDATE outbound_queue SET status = 'UNKNOWN_ERROR', last_error = ?, updated_at = ? "
                    "WHERE task_id = ? AND worker_id = ? AND claim_token = ? AND status = 'SENDING'",
                    (f"弱网/异常无回执，等待对账: {error_detail}", now, task_id, worker_id, claim_token)
                )
                if cur.rowcount == 1:
                    await db.execute("COMMIT")
                    logger.critical(f"[出站结果不确定] 任务 {task_id} 结果未知，已置为 UNKNOWN_ERROR！")
                    return
            await db.execute("ROLLBACK")
        except Exception as e:
            await db.execute("ROLLBACK")
            logger.error(f"[出站结果落库异常] {e}")
```

### 2. 修复 P1-01：按钮执行增加未知异常防护（严禁未知状态直接允许盲目重试）

```python
async def execute_button_action_atomically(
    db, 
    write_lock: asyncio.Lock, 
    action_id: str, 
    operator_id: int, 
    chat_id: int, 
    execute_coro
) -> bool:
    """
    【事务 6.1 修订：按钮执行与幂等状态机】
    【P1-01 修复】：区分“确定性执行失败”与“网络超时/未知异常”，未知状态置 UNKNOWN_ERROR，严禁直接允许重复操作
    """
    now = int(time.time())
    execution_token = secrets.token_hex(16)
    execution_expires_at = now + 60  # 60 秒执行租约
    
    # 1. 原子抢占状态 PENDING/FAILED -> EXECUTING
    async with write_lock:
        await db.execute("BEGIN IMMEDIATE")
        cur = await db.execute(
            "UPDATE button_actions SET status = 'EXECUTING', operator_id = ?, executing_at = ?, "
            "execution_expires_at = ?, execution_token = ?, last_error = NULL "
            "WHERE action_id = ? AND status IN ('PENDING', 'FAILED') AND expires_at > ?",
            (operator_id, now, execution_expires_at, execution_token, action_id, now)
        )
        if cur.rowcount != 1:
            await db.execute("ROLLBACK")
            return False
        
        cur = await db.execute("SELECT action_type, action_value FROM button_actions WHERE action_id = ?", (action_id,))
        action_type, action_value = await cur.fetchone()
        await db.execute("COMMIT")

    # 2. 外部 API 调用 (在锁外执行网络 I/O，规避长事务)
    success = False
    is_indeterminate = False
    error_msg = None
    try:
        await execute_coro(action_type, action_value)
        success = True
    except DETERMINISTIC_FAILED_EXCEPTIONS as det_err:
        # 确定性失败（权限不足、用户不存在等，明确未产生副作用，允许重试）
        error_msg = str(det_err)
    except Exception as e:
        # 超时、网络断开等未知异常（可能已封禁成功但回执丢了，禁止盲目重试！）
        is_indeterminate = True
        error_msg = f"外部执行结果不确定: {e}"
        logger.error(f"[按钮执行未知异常] {e}")

    # 3. 最终落库与本地审计
    now_finish = int(time.time())
    async with write_lock:
        await db.execute("BEGIN IMMEDIATE")
        try:
            if success:
                target_status = 'COMPLETED'
            elif is_indeterminate:
                target_status = 'UNKNOWN_ERROR'  # 锁死按钮，等待人工确认，防止二次副作用
            else:
                target_status = 'FAILED'         # 允许再次点击重试

            cur = await db.execute(
                f"UPDATE button_actions SET status = ?, executed_at = ?, last_error = ? "
                f"WHERE action_id = ? AND execution_token = ? AND status = 'EXECUTING'",
                (target_status, now_finish, error_msg, action_id, execution_token)
            )
            
            if cur.rowcount != 1:
                await db.execute("ROLLBACK")
                logger.error(f"[按钮状态完成异常] action_id={action_id} rowcount!=1，事务回滚！")
                return False

            # 写入本地操作审计日志
            await db.execute(
                "INSERT INTO admin_audit_logs (action_id, operator_id, chat_id, action_type, action_value, result, detail, created_at) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                (action_id, operator_id, chat_id, action_type, action_value, target_status, error_msg, now_finish)
            )
            await db.execute("COMMIT")
            return success
        except Exception as e:
            await db.execute("ROLLBACK")
            logger.error(f"[按钮提交事务异常] {e}")
            return False

```

### 3. 修复 P1-03：在线备份唯一文件名与独立恢复前后数据行数比对核验

```python
import os
import time
import shutil
import aiosqlite
import logging

logger = logging.getLogger("tg-monitor-backup")

async def perform_online_backup(db_path: str, backup_dir: str, keep_days: int = 7) -> str | None:
    """
    【在线备份修订】
    【P1-03 修复】：采用时间戳+唯一随机后缀，绝对杜绝文件名冲突导致误删既有快照
    """
    os.makedirs(backup_dir, exist_ok=True)
    # 秒级时间戳 + 8位唯一随机 Token，防止并发同秒冲突
    unique_suffix = f"{time.strftime('%Y%m%d_%H%M%S')}_{secrets.token_hex(4)}"
    backup_file = os.path.join(backup_dir, f"tg_monitor_backup_{unique_suffix}.db")

    try:
        if not os.path.exists(db_path):
            raise FileNotFoundError(f"源数据库不存在: {db_path}")

        async with aiosqlite.connect(db_path) as db:
            await db.execute(f"VACUUM INTO '{backup_file}'")

        if not os.path.exists(backup_file) or os.path.getsize(backup_file) == 0:
            raise IOError("备份文件生成失败或体积为0字节")

        async with aiosqlite.connect(backup_file) as b_db:
            cur = await b_db.execute("PRAGMA integrity_check")
            result = (await cur.fetchone())[0]
            if result != "ok":
                raise ValueError(f"备份文件完整性损坏: {result}")

        logger.info(f"[数据库备份成功] 生成快照: {backup_file} (体积: {os.path.getsize(backup_file)} 字节)")

        # 历史快照安全轮转
        now_ts = time.time()
        for f in os.listdir(backup_dir):
            if f.startswith("tg_monitor_backup_") and f.endswith(".db"):
                f_path = os.path.join(backup_dir, f)
                if now_ts - os.path.getmtime(f_path) > (keep_days * 86400):
                    os.remove(f_path)
                    logger.info(f"[历史备份清理] 已淘汰过期备份: {f}")

        return backup_file
    except Exception as e:
        logger.error(f"[数据库备份失败] {e}")
        # 仅安全删除本次实际尝试创建的文件，绝不误删旧备份
        if os.path.exists(backup_file):
            os.remove(backup_file)
        return None

async def execute_isolated_restore_drill(source_db_path: str, backup_file: str, drill_dir: str) -> bool:
    """
    【独立恢复演练修订】
    【P1-03 修复】：采集源库 7 表实际记录数，恢复到独立沙箱目录后逐表行数对比，确保 100% 数据一致
    """
    os.makedirs(drill_dir, exist_ok=True)
    restored_db_path = os.path.join(drill_dir, "restored_drill.db")
    required_tables = [
        "message_dedup", "inbound_queue", "user_cooldown", 
        "outbound_queue", "button_actions", "runtime_state", "admin_audit_logs"
    ]

    try:
        # 1. 采集备份源库的基准表行数
        source_counts = {}
        async with aiosqlite.connect(source_db_path) as s_db:
            for tbl in required_tables:
                cur = await s_db.execute(f"SELECT COUNT(*) FROM {tbl}")
                source_counts[tbl] = (await cur.fetchone())[0]

        # 2. 拷贝备份文件到演练沙箱目录
        shutil.copyfile(backup_file, restored_db_path)

        # 3. 校验沙箱库完整性并逐表对比行数
        async with aiosqlite.connect(restored_db_path) as r_db:
            cur = await r_db.execute("PRAGMA integrity_check")
            if (await cur.fetchone())[0] != "ok":
                logger.error("[恢复演练失败] PRAGMA integrity_check 未通过")
                return False

            for tbl in required_tables:
                cur = await r_db.execute(f"SELECT COUNT(*) FROM {tbl}")
                restored_count = (await cur.fetchone())[0]
                if restored_count != source_counts[tbl]:
                    logger.error(f"[恢复演练失败] 表 {tbl} 行数不一致: 源库 {source_counts[tbl]} != 恢复库 {restored_count}")
                    return False
                logger.debug(f"[恢复演练核验] 表 {tbl} 数据一致 (行数: {restored_count})")

        logger.info(f"[恢复演练通过] 备份文件 {backup_file} 数据行数与业务记录 100% 吻合！")
        return True
    except Exception as e:
        logger.error(f"[恢复演练异常] {e}")
        return False
    finally:
        if os.path.exists(restored_db_path):
            os.remove(restored_db_path)
```

### 4. 修复 P1-04：数据清理白名单保护与消除不严谨安全表述

```python
async def perform_scheduled_data_cleanup(db, write_lock: asyncio.Lock):
    """
    【数据清理修订】
    【P1-04 修复】：清理严格限定在终态白名单，绝对排除 EXECUTING 运行态，且严禁触碰审计日志
    """
    now = int(time.time())
    seven_days_ago = now - (7 * 86400)
    three_days_ago = now - (3 * 86400)
    one_day_ago = now - 86400

    async with write_lock:
        await db.execute("BEGIN IMMEDIATE")
        try:
            # 1. 消息去重表：清理 7 天前记录
            cur = await db.execute("DELETE FROM message_dedup WHERE created_at < ?", (seven_days_ago,))
            del_dedup = cur.rowcount

            # 2. 入站任务表：仅清理 3 天前的 COMPLETED 与 DROPPED_COOLDOWN (排除 FAILED)
            cur = await db.execute(
                "DELETE FROM inbound_queue WHERE status IN ('COMPLETED', 'DROPPED_COOLDOWN') AND updated_at < ?", 
                (three_days_ago,)
            )
            del_inbound = cur.rowcount

            # 3. 出站任务表：仅清理 7 天前已成功送达的 SENT (排除 MANUAL_REVIEW / DEAD / UNKNOWN_ERROR)
            cur = await db.execute("DELETE FROM outbound_queue WHERE status = 'SENT' AND updated_at < ?", (seven_days_ago,))
            del_outbound = cur.rowcount

            # 4. 按钮交互表【修正】：明确限定为 COMPLETED 终态，绝对排除 EXECUTING 与 UNKNOWN_ERROR
            cur = await db.execute(
                "DELETE FROM button_actions WHERE status = 'COMPLETED' AND expires_at < ? AND created_at < ?",
                (now, one_day_ago)
            )
            del_btn = cur.rowcount

            await db.execute("COMMIT")
            logger.info(
                f"[数据生命周期清理完成] 去重记录: {del_dedup}, 入站任务: {del_inbound}, "
                f"出站任务: {del_outbound}, 按钮记录: {del_btn} (所有故障记录与本地操作审计日志均完好保留)"
            )
        except Exception as e:
            await db.execute("ROLLBACK")
            logger.error(f"[数据生命周期清理异常] {e}")
```
